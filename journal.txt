    async function callApi(payload) {
      try {
        const resPost = await fetch(SCRIPT_API_URL, {
          method: "POST",
          headers: { "Content-Type": "text/plain;charset=utf-8" },
          body: JSON.stringify(payload)
        });
        const text = await resPost.text();
        return JSON.parse(text);
      } catch (err) {
        const qs = new URLSearchParams();
        Object.keys(payload).forEach(k => {
          if (payload[k] !== undefined && payload[k] !== null && typeof payload[k] !== 'object') {
            qs.append(k, String(payload[k]));
          }
        });
        const resGet = await fetch(`${SCRIPT_API_URL}?${qs.toString()}`);
        return await resGet.json();
      }
    }

    function maskIdDisplay(rawStr) {
      const s = String(rawStr || "").trim();
      if (!s) return "VIP*****";
      if (s.includes("*")) return s;
      if (s.toUpperCase().startsWith("VIP")) return "VIP****" + s.slice(-2);
      if (s.length <= 4) return s.charAt(0) + "****" + s.slice(-1);
      return s.slice(0, 2) + "*****" + s.slice(-2);
    }

    function calculateAutoFee(side, entryPrice, exitPrice, quantity, netProfit) {
      const ep = parseFloat(entryPrice) || 0;
      const xp = parseFloat(exitPrice) || 0;
      const qty = parseFloat(quantity) || 0;
      const net = parseFloat(netProfit) || 0;
      if (ep <= 0 || xp <= 0 || qty <= 0) return 0;

      const isShort = String(side || "").includes("空") || String(side || "").toLowerCase().includes("short");
      const grossPnl = isShort ? (ep - xp) * qty : (xp - ep) * qty;
      const diff = grossPnl - net;

      if (Math.abs(diff) >= 0.01) {
        return Math.round(Math.abs(diff) * 100) / 100;
      }
      const turnover = (ep * qty) + (xp * qty);
      return Math.round((turnover * 0.0005) * 100) / 100;
    }

      return fee;
    }

    function renderStudentTrades() {
      const container = document.getElementById('studentTradesContainer');
      const initCap = parseFloat(document.getElementById('stuInitialCapitalInput').value) || 1000;
      let totalPnl = 0, totalFee = 0, wins = 0;

      studentTrades.forEach(t => {
        const pnl = parseFloat(t.netProfit) || 0;
        const qty = parseFloat(t.quantity || t.margin) || 0;
        const computedFee = (t.fee && parseFloat(t.fee) > 0)
          ? parseFloat(t.fee)
          : calculateAutoFee(t.side, t.entryPrice, t.exitPrice, qty, pnl);
        t._computedFee = computedFee;
        totalPnl += pnl;
        totalFee += computedFee;
        if (pnl > 0) wins++;
      });

      const count = studentTrades.length;
      const winRate = count > 0 ? ((wins / count) * 100).toFixed(1) : "0.0";
      const roi = initCap > 0 ? ((totalPnl / initCap) * 100).toFixed(1) : "0.0";

      document.getElementById('statCurrentEquity').textContent = `$${(initCap + totalPnl).toFixed(2)}`;
      document.getElementById('statTotalPnl').textContent = `${totalPnl >= 0 ? '+' : ''}$${totalPnl.toFixed(2)}`;
      document.getElementById('statTotalRoi').textContent = `${totalPnl >= 0 ? '+' : ''}${roi}%`;
      document.getElementById('statWinRate').textContent = `${winRate}%`;
      document.getElementById('statTotalFee').textContent = `$${totalFee.toFixed(2)}`;
      document.getElementById('statTradeCount').textContent = `${count} 筆`;

      if (count === 0) {
        container.innerHTML = `<div class="text-center py-8 theme-text-muted text-sm">目前尚無您個人的交易紀錄。</div>`;
        return;
      }

      container.innerHTML = studentTrades.map((t, idx) => {
        const pnl = parseFloat(t.netProfit) || 0;
        return `
          <div class="p-4 rounded-2xl theme-sub">
            <div class="flex flex-wrap items-center justify-between gap-2 mb-2">
              <div class="flex items-center gap-2">
                <span class="font-black theme-text-main">${t.symbol}</span>
                <span class="text-xs px-2 py-0.5 rounded bg-amber-500/15 text-amber-500 amber-badge-text font-bold">${t.side}${t.leverage}x</span>

                <span class="px-3 py-1 rounded-lg font-black text-xs ${pnl >= 0 ? 'bg-emerald-500/20 text-emerald-500' : 'bg-rose-500/20 text-rose-500'}">${pnl >= 0 ? '+' : ''}${pnl} U</span>
                <button type="button" onclick="editTrade(${idx})" class="px-2.5 py-1 rounded-lg theme-card text-xs theme-text-main font-bold">修改</button>
                <button type="button" onclick="deleteTrade(${idx})" class="px-2.5 py-1 rounded-lg bg-rose-500/15 text-xs text-rose-500 font-bold">刪除</button>
              </div>
            </div>

              <span>開倉均價: <strong class="theme-text-main">${t.entryPrice}</strong></span>
              <span>平倉均價: <strong class="theme-text-main">${t.exitPrice}</strong></span>

    function renderStudentResources() {
      const lecturesContainer = document.getElementById('studentLecturesContainer');
      const websitesContainer = document.getElementById('studentWebsitesContainer');
      const kw = (document.getElementById('resSearchInput')?.value || "").trim().toLowerCase();
      const resources = Array.isArray(window.globalResources) ? window.globalResources : [];

      const lectures = resources.filter(item => String(item.type).includes('講義'));
      const websites = resources.filter(item => !String(item.type).includes('講義') && (
        !kw || String(item.title).toLowerCase().includes(kw) || String(item.note).toLowerCase().includes(kw) || String(item.tier).toLowerCase().includes(kw)
      ));

      if (lecturesContainer) {
        lecturesContainer.innerHTML = lectures.map((item, idx) => `
          <a href="${item.url || '#'}" target="_blank" rel="noopener noreferrer" class="block p-4 rounded-xl theme-sub hover:border-amber-500 transition-all">

            <h4 class="font-bold theme-text-main text-sm mb-1">${item.title}</h4>
            <p class="text-xs theme-text-muted">${item.note || ''}</p>
          </a>
        `).join('');
      }

      if (websitesContainer) {
        const tierGroups = {};
        const tierOrder = [];
        websites.forEach(site => {
          const tName = (site.tier && String(site.tier).trim()) ? String(site.tier).trim() : '🏆 第一梯隊：即時看盤、爆倉清算與多空籌碼';
          if (!tierGroups[tName]) { tierGroups[tName] = []; tierOrder.push(tName); }
          tierGroups[tName].push(site);
        });

        websitesContainer.innerHTML = tierOrder.map(tierName => {
          const groupSites = tierGroups[tierName];
          const cards = groupSites.map((site, i) => `
            <a href="${site.url || '#'}" target="_blank" rel="noopener noreferrer" class="p-4 rounded-xl theme-card hover:border-cyan-500 transition-all block">

            </a>

            <div class="theme-sub rounded-2xl p-4 sm:p-5">
              <div class="flex items-center justify-between border-b border-slate-700/40 pb-2.5 mb-3.5">
                <h4 class="text-base font-extrabold text-amber-500 amber-badge-text">${tierName}</h4>

            </div>
          `;
        }).join('');
      }
    }

    function runRiskCalc() {
      const cap = parseFloat(document.getElementById('calcCapital').value) || 0;
      const riskPct = parseFloat(document.getElementById('calcRiskPct').value) || 0;
      const entry = parseFloat(document.getElementById('calcEntry').value) || 0;
      const sl = parseFloat(document.getElementById('calcSl').value) || 0;
      const tp = parseFloat(document.getElementById('calcTp').value) || 0;
      const lev = parseFloat(document.getElementById('calcLev').value) || 10;

      const riskUsd = cap * (riskPct / 100);
      document.getElementById('resRiskUsd').textContent = `$${riskUsd.toFixed(2)} U`;
      if (entry > 0 && sl > 0 && entry !== sl) {
        const posVal = riskUsd / (Math.abs(entry - sl) / entry);
        document.getElementById('resPosValue').textContent = `$${posVal.toFixed(2)} U`;
        document.getElementById('resMarginNeed').textContent = `$${(posVal / lev).toFixed(2)} U`;
        document.getElementById('resRR').textContent = tp > 0 ? `1 : ${(Math.abs(tp - entry) / Math.abs(entry - sl)).toFixed(2)}` : "--";
      }
    }

    function runPyramidCalc() {
      let totalQ = 0, totalVal = 0;
      document.querySelectorAll('.pyr-row').forEach(r => {
        const p = parseFloat(r.querySelector('.pyr-price')?.value) || 0;
        const q = parseFloat(r.querySelector('.pyr-qty')?.value) || 0;
        if (p > 0 && q > 0) {
          totalQ += q;
          totalVal += (p * q);
        }
      });
      const avg = totalQ > 0 ? (totalVal / totalQ) : 0;
      document.getElementById('pyrAvgPrice').textContent = avg > 0 ? avg.toFixed(4) : "0.00";
      document.getElementById('pyrTotalQty').textContent = totalQ > 0 ? totalQ.toFixed(4) : "0.00";
      document.getElementById('pyrTotalVal').textContent = `$${totalVal.toFixed(2)} U`;
    }

    function renderLeaderboard() {
      const container = document.getElementById('leaderboardContainer');
      const capProfit = parseFloat(maxLimits.maxProfit) || 4850;
      const capVolume = parseFloat(maxLimits.maxVolume) || 185000;

      const rawList = (globalLeaderboard.length > 0 ? [...globalLeaderboard] : [
        { nickname: "VIP****88", winRate: 78.5, netProfit: capProfit, volume: capVolume },
        { nickname: "83*****19", winRate: 74.2, netProfit: Math.round(capProfit * 0.78), volume: Math.round(capVolume * 0.82) },
        { nickname: "VIP****06", winRate: 71.0, netProfit: Math.round(capProfit * 0.62), volume: Math.round(capVolume * 0.65) }
      ]).map(item => ({
        nickname: maskIdDisplay(item.nickname),
        winRate: item.winRate || 68.0,
        netProfit: Math.min(parseFloat(item.netProfit) || 0, capProfit),
        volume: Math.min(parseFloat(item.volume) || 0, capVolume)
      }));

      if (currentLbMode === 'volume') {
        rawList.sort((a, b) => (b.volume || 0) - (a.volume || 0));

        rawList.sort((a, b) => (b.netProfit || 0) - (a.netProfit || 0));

      container.innerHTML = rawList.slice(0, 10).map((item, i) => {
        const rightText = currentLbMode === 'volume'
          ? `<span class="font-black text-cyan-500 cyan-badge-text text-sm">交易量 $${Math.round(item.volume || 0).toLocaleString()} U</span>`
          : `<span class="font-black text-emerald-500 text-sm">盈利 +$${Math.round(item.netProfit || 0).toLocaleString()} U</span>`;
        return `
          <div class="flex items-center justify-between p-3.5 rounded-xl theme-sub">
            <div class="flex items-center gap-3">
              <span class="font-black text-amber-500 amber-badge-text w-6">#${i + 1}</span>
              <span class="font-bold theme-text-main text-sm font-mono">${item.nickname} <span class="text-xs theme-text-muted font-sans ml-1">(勝率 ${item.winRate}%)</span></span>
            </div>
            ${rightText}
          </div>
        `;
      }).join('');
    }

    function renderAdminStudents() {

      if (!container) return;

      if (allStudentsAdmin.length === 0) {
        container.innerHTML = `<div class="text-xs theme-text-muted py-4">尚無學員申請資料（可點上方「🔄 雲端同步」拉取最新名單）。</div>`;
        return;
      }

      const defaultSubDate = new Date(Date.now() + 30 * 86400000).toISOString().split('T')[0];

        const isSuspended = (stu.status === "已暫停" || stu.status === "暫停");

    async function saveStudentRow(rowIndex, convertMode) {
      if (!currentUser || currentUser.role !== 'admin') return;
      const statusVal = document.getElementById(`admStatus_${rowIndex}`)?.value || "已開通";
      const accVal = document.getElementById(`admAcc_${rowIndex}`)?.value.trim() || "";
      const pwdVal = document.getElementById(`admPwd_${rowIndex}`)?.value.trim() || "";
      const expVal = document.getElementById(`admExp_${rowIndex}`)?.value || "";

      showSyncToast("⏳ 正在儲存學員帳密並同步更新私訊文案...");
      try {
        const data = await callApi({
          action: "adminUpdateStudent",
          rowIndex: rowIndex,
          newStatus: convertMode ? "已開通" : statusVal,
          customAccount: accVal,
          customPassword: pwdVal,
          customExpireDate: expVal,
          convertMode: convertMode || ""
        });
        if (data && Array.isArray(data.students)) {
          allStudentsAdmin = data.students;
          saveSessionCache();
          renderAdminStudents();
        }
        showSyncToast("✅ 已更新學員設定與私訊發送文案！", 1800);
      } catch (err) {
        hideSyncToast();
        alert("更新失敗：" + err.message);
      }
    }
